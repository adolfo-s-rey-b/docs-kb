This example showcases a simple horizontal bar chart.

import matplotlib.pyplot as plt

[fig](https://matplotlib.org/stable/api/_as_gen/matplotlib.figure.Figure.html#matplotlib.figure.Figure "matplotlib.figure.Figure")
, [ax](https://matplotlib.org/stable/api/_as_gen/matplotlib.axes.Axes.html#matplotlib.axes.Axes "matplotlib.axes.Axes")
 \= [plt.subplots](https://matplotlib.org/stable/api/_as_gen/matplotlib.pyplot.subplots.html#matplotlib.pyplot.subplots "matplotlib.pyplot.subplots")
()

\# Example data
[people](https://docs.python.org/3/library/stdtypes.html#tuple "builtins.tuple")
 \= ('Tom', 'Dick', 'Harry', 'Slim', 'Jim')
[performance](https://docs.python.org/3/library/stdtypes.html#list "builtins.list")
 \= \[5, 7, 6, 4, 9\]
[error](https://docs.python.org/3/library/stdtypes.html#list "builtins.list")
 \= \[0.2, 0.4, 0.3, 0.6, 0.2\]

[ax.barh](https://matplotlib.org/stable/api/_as_gen/matplotlib.axes.Axes.barh.html#matplotlib.axes.Axes.barh "matplotlib.axes.Axes.barh")
([people](https://docs.python.org/3/library/stdtypes.html#tuple "builtins.tuple")
, [performance](https://docs.python.org/3/library/stdtypes.html#list "builtins.list")
, xerr\=[error](https://docs.python.org/3/library/stdtypes.html#list "builtins.list")
, align\='center')
[ax.yaxis.set\_inverted](https://matplotlib.org/stable/api/_as_gen/matplotlib.axis.Axis.set_inverted.html#matplotlib.axis.Axis.set_inverted "matplotlib.axis.Axis.set_inverted")
(True)  \# arrange data from top to bottom
[ax.set\_xlabel](https://matplotlib.org/stable/api/_as_gen/matplotlib.axes.Axes.set_xlabel.html#matplotlib.axes.Axes.set_xlabel "matplotlib.axes.Axes.set_xlabel")
('Performance')
[ax.set\_title](https://matplotlib.org/stable/api/_as_gen/matplotlib.axes.Axes.set_title.html#matplotlib.axes.Axes.set_title "matplotlib.axes.Axes.set_title")
('How fast do you want to go today?')

[plt.show](https://matplotlib.org/stable/api/_as_gen/matplotlib.pyplot.show.html#matplotlib.pyplot.show "matplotlib.pyplot.show")
()
