import matplotlib.pyplot as plt
import numpy as np

plt.style.use('\_mpl-gallery')

\# Make data
[x](https://docs.python.org/3/library/stdtypes.html#list "builtins.list")
 \= \[1, 1, 2, 2\]
[y](https://docs.python.org/3/library/stdtypes.html#list "builtins.list")
 \= \[1, 2, 1, 2\]
[z](https://docs.python.org/3/library/stdtypes.html#list "builtins.list")
 \= \[0, 0, 0, 0\]
[dx](https://numpy.org/doc/stable/reference/generated/numpy.ndarray.html#numpy.ndarray "numpy.ndarray")
 \= [np.ones\_like](https://numpy.org/doc/stable/reference/generated/numpy.ones_like.html#numpy.ones_like "numpy.ones_like")
([x](https://docs.python.org/3/library/stdtypes.html#list "builtins.list")
)\*0.5
[dy](https://numpy.org/doc/stable/reference/generated/numpy.ndarray.html#numpy.ndarray "numpy.ndarray")
 \= [np.ones\_like](https://numpy.org/doc/stable/reference/generated/numpy.ones_like.html#numpy.ones_like "numpy.ones_like")
([x](https://docs.python.org/3/library/stdtypes.html#list "builtins.list")
)\*0.5
[dz](https://docs.python.org/3/library/stdtypes.html#list "builtins.list")
 \= \[2, 3, 1, 4\]
