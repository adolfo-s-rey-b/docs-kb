This example demonstrates how to set a hyperlinks on various kinds of elements.

This currently only works with the SVG backend.

import matplotlib.pyplot as plt
import numpy as np

fig \= [plt.figure](https://matplotlib.org/stable/api/_as_gen/matplotlib.pyplot.figure.html#matplotlib.pyplot.figure "matplotlib.pyplot.figure")
()
s \= [plt.scatter](https://matplotlib.org/stable/api/_as_gen/matplotlib.pyplot.scatter.html#matplotlib.pyplot.scatter "matplotlib.pyplot.scatter")
(\[1, 2, 3\], \[4, 5, 6\])
s.set\_urls(\['https://www.bbc.com/news', 'https://www.google.com/', None\])
fig.savefig('scatter.svg')

fig \= [plt.figure](https://matplotlib.org/stable/api/_as_gen/matplotlib.pyplot.figure.html#matplotlib.pyplot.figure "matplotlib.pyplot.figure")
()
delta \= 0.025
x \= y \= [np.arange](https://numpy.org/doc/stable/reference/generated/numpy.arange.html#numpy.arange "numpy.arange")
(\-3.0, 3.0, delta)
X, Y \= [np.meshgrid](https://numpy.org/doc/stable/reference/generated/numpy.meshgrid.html#numpy.meshgrid "numpy.meshgrid")
(x, y)
Z1 \= [np.exp](https://numpy.org/doc/stable/reference/generated/numpy.exp.html#numpy.exp "numpy.exp")
(\-X\*\*2 \- Y\*\*2)
Z2 \= [np.exp](https://numpy.org/doc/stable/reference/generated/numpy.exp.html#numpy.exp "numpy.exp")
(\-(X \- 1)\*\*2 \- (Y \- 1)\*\*2)
Z \= (Z1 \- Z2) \* 2

im \= [plt.imshow](https://matplotlib.org/stable/api/_as_gen/matplotlib.pyplot.imshow.html#matplotlib.pyplot.imshow "matplotlib.pyplot.imshow")
(Z, interpolation\='bilinear', cmap\="gray",
                origin\='lower', extent\=(\-3, 3, \-3, 3))

im.set\_url('https://www.google.com/')
fig.savefig('image.svg')
