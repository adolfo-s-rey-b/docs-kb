Hinton diagrams[#](https://matplotlib.org/stable/gallery/specialty_plots/hinton_demo.html#hinton-diagrams "Link to this heading")

==================================================================================================================================

Hinton diagrams are useful for visualizing the values of a 2D array (e.g. a weight matrix): Positive and negative values are represented by white and black squares, respectively, and the size of each square represents the magnitude of each value.

Initial idea from David Warde-Farley on the SciPy Cookbook

![hinton demo](https://matplotlib.org/stable/_images/sphx_glr_hinton_demo_001_2_00x.png)

import numpy as np

def hinton(matrix, max\_weight\=None, ax\=None):
    """Draw Hinton diagram for visualizing a weight matrix."""
    ax \= ax if ax is not None else [plt.gca](https://matplotlib.org/stable/api/_as_gen/matplotlib.pyplot.gca.html#matplotlib.pyplot.gca "matplotlib.pyplot.gca")
()

    if not max\_weight:
        max\_weight \= 2 \*\* [np.ceil](https://numpy.org/doc/stable/reference/generated/numpy.ufunc.html#numpy.ufunc "numpy.ufunc")
([np.log2](https://numpy.org/doc/stable/reference/generated/numpy.ufunc.html#numpy.ufunc "numpy.ufunc")
([np.abs](https://numpy.org/doc/stable/reference/generated/numpy.ufunc.html#numpy.ufunc "numpy.ufunc")
(matrix).max()))

    ax.patch.set\_facecolor('gray')
    ax.set\_aspect('equal', 'box')
    ax.xaxis.set\_major\_locator([plt.NullLocator](https://matplotlib.org/stable/api/ticker_api.html#matplotlib.ticker.NullLocator "matplotlib.ticker.NullLocator")
())
    ax.yaxis.set\_major\_locator([plt.NullLocator](https://matplotlib.org/stable/api/ticker_api.html#matplotlib.ticker.NullLocator "matplotlib.ticker.NullLocator")
())

    for (x, y), w in [np.ndenumerate](https://numpy.org/doc/stable/reference/generated/numpy.ndenumerate.html#numpy.ndenumerate "numpy.ndenumerate")

        color \= 'white' if w \> 0 else 'black'
        size \= [np.sqrt](https://numpy.org/doc/stable/reference/generated/numpy.ufunc.html#numpy.ufunc "numpy.ufunc")
(abs(w) / max\_weight)
        rect \= [plt.Rectangle](https://matplotlib.org/stable/api/_as_gen/matplotlib.patches.Rectangle.html#matplotlib.patches.Rectangle "matplotlib.patches.Rectangle")
(\[x \- size / 2, y \- size / 2\], size, size,
                             facecolor\=color, edgecolor\=color)
        ax.add\_patch(rect)

    ax.autoscale\_view()
    ax.invert\_yaxis()
