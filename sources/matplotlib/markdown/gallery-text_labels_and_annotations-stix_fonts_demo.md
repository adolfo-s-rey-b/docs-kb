========================================================================================================================================

Demonstration of [STIX Fonts](https://www.stixfonts.org/)
 used in LaTeX rendering.

![stix fonts demo](https://matplotlib.org/stable/_images/sphx_glr_stix_fonts_demo_001_2_00x.png)

import matplotlib.pyplot as plt

[circle123](https://docs.python.org/3/library/stdtypes.html#str "builtins.str")
 \= "\\N{CIRCLED DIGIT ONE}\\N{CIRCLED DIGIT TWO}\\N{CIRCLED DIGIT THREE}"

[tests](https://docs.python.org/3/library/stdtypes.html#list "builtins.list")
 \= \[\
    r'$%s\\;\\mathrm{%s}\\;\\mathbf{%s}$' % (([circle123](https://docs.python.org/3/library/stdtypes.html#str "builtins.str")\
,) \* 3),\
    r'$\\mathsf{Sans \\Omega}\\;\\mathrm{\\mathsf{Sans \\Omega}}\\;'\
    r'\\mathbf{\\mathsf{Sans \\Omega}}$',\
    r'$\\mathtt{Monospace}$',\
    r'$\\mathcal{CALLIGRAPHIC}$',\
    r'$\\mathbb{Blackboard\\;\\pi}$',\
    r'$\\mathrm{\\mathbb{Blackboard\\;\\pi}}$',\
    r'$\\mathbf{\\mathbb{Blackboard\\;\\pi}}$',\
    r'$\\mathfrak{Fraktur}\\;\\mathbf{\\mathfrak{Fraktur}}$',\
    r'$\\mathscr{Script}$',\
\]

[fig](https://matplotlib.org/stable/api/_as_gen/matplotlib.figure.Figure.html#matplotlib.figure.Figure "matplotlib.figure.Figure")
 \= [plt.figure](https://matplotlib.org/stable/api/_as_gen/matplotlib.pyplot.figure.html#matplotlib.pyplot.figure "matplotlib.pyplot.figure")
(figsize\=(8, len([tests](https://docs.python.org/3/library/stdtypes.html#list "builtins.list")
) + 2))
for [i](https://docs.python.org/3/library/functions.html#int "builtins.int")
, [s](https://docs.python.org/3/library/stdtypes.html#str "builtins.str")
 in enumerate([tests](https://docs.python.org/3/library/stdtypes.html#list "builtins.list")
\[::\-1\]):
    [fig.text](https://matplotlib.org/stable/api/_as_gen/matplotlib.figure.Figure.text.html#matplotlib.figure.Figure.text "matplotlib.figure.Figure.text")
(0, ([i](https://docs.python.org/3/library/functions.html#int "builtins.int")
 + .5) / len([tests](https://docs.python.org/3/library/stdtypes.html#list "builtins.list")
), [s](https://docs.python.org/3/library/stdtypes.html#str "builtins.str")
, fontsize\=32)

[plt.show](https://matplotlib.org/stable/api/_as_gen/matplotlib.pyplot.show.html#matplotlib.pyplot.show "matplotlib.pyplot.show")
()
