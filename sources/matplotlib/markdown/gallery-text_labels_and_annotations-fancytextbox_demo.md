==========================================================================================================================================================

This example shows how to style text boxes using _bbox_ parameters.

![fancytextbox demo](https://matplotlib.org/stable/_images/sphx_glr_fancytextbox_demo_001_2_00x.png)

import matplotlib.pyplot as plt

[plt.text](https://matplotlib.org/stable/api/_as_gen/matplotlib.pyplot.text.html#matplotlib.pyplot.text "matplotlib.pyplot.text")
(0.6, 0.7, "eggs", size\=50, rotation\=30.,
         ha\="center", va\="center",
         bbox\=dict(boxstyle\="round",
                   ec\=(1., 0.5, 0.5),
                   fc\=(1., 0.8, 0.8),
                   )
         )

[plt.text](https://matplotlib.org/stable/api/_as_gen/matplotlib.pyplot.text.html#matplotlib.pyplot.text "matplotlib.pyplot.text")
(0.55, 0.6, "spam", size\=50, rotation\=-25.,
         ha\="right", va\="top",
         bbox\=dict(boxstyle\="square",
                   ec\=(1., 0.5, 0.5),
                   fc\=(1., 0.8, 0.8),
                   )
         )

[plt.show](https://matplotlib.org/stable/api/_as_gen/matplotlib.pyplot.show.html#matplotlib.pyplot.show "matplotlib.pyplot.show")
()
