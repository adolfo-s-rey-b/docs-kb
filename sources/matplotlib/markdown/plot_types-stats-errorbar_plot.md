errorbar(x, y, yerr, xerr)[#](https://matplotlib.org/stable/plot_types/stats/errorbar_plot.html#errorbar-x-y-yerr-xerr "Link to this heading")

===============================================================================================================================================

Plot y versus x as lines and/or markers with attached errorbars.

See [`errorbar`](https://matplotlib.org/stable/api/_as_gen/matplotlib.axes.Axes.errorbar.html#matplotlib.axes.Axes.errorbar "matplotlib.axes.Axes.errorbar")
.

![errorbar plot](https://matplotlib.org/stable/_images/sphx_glr_errorbar_plot_001_2_00x.png)

import matplotlib.pyplot as plt
import numpy as np

plt.style.use('\_mpl-gallery')

\# make data:
[np.random.seed](https://numpy.org/doc/stable/reference/random/generated/numpy.random.seed.html#numpy.random.seed "numpy.random.seed")
(1)
[x](https://docs.python.org/3/library/stdtypes.html#list "builtins.list")
 \= \[2, 4, 6\]
[y](https://docs.python.org/3/library/stdtypes.html#list "builtins.list")
 \= \[3.6, 5, 4.2\]
[yerr](https://docs.python.org/3/library/stdtypes.html#list "builtins.list")
 \= \[0.9, 1.2, 0.5\]

\# plot:
[fig](https://matplotlib.org/stable/api/_as_gen/matplotlib.figure.Figure.html#matplotlib.figure.Figure "matplotlib.figure.Figure")
, [ax](https://matplotlib.org/stable/api/_as_gen/matplotlib.axes.Axes.html#matplotlib.axes.Axes "matplotlib.axes.Axes")
 \= [plt.subplots](https://matplotlib.org/stable/api/_as_gen/matplotlib.pyplot.subplots.html#matplotlib.pyplot.subplots "matplotlib.pyplot.subplots")
()

[ax.errorbar](https://matplotlib.org/stable/api/_as_gen/matplotlib.axes.Axes.errorbar.html#matplotlib.axes.Axes.errorbar "matplotlib.axes.Axes.errorbar")
([x](https://docs.python.org/3/library/stdtypes.html#list "builtins.list")
, [y](https://docs.python.org/3/library/stdtypes.html#list "builtins.list")
, [yerr](https://docs.python.org/3/library/stdtypes.html#list "builtins.list")
, fmt\='o', linewidth\=2, capsize\=6)

[ax.set](https://matplotlib.org/stable/api/_as_gen/matplotlib.axes.Axes.set.html#matplotlib.axes.Axes.set "matplotlib.axes.Axes.set")
(xlim\=(0, 8), xticks\=[np.arange](https://numpy.org/doc/stable/reference/generated/numpy.arange.html#numpy.arange "numpy.arange")
(1, 8),
       ylim\=(0, 8), yticks\=[np.arange](https://numpy.org/doc/stable/reference/generated/numpy.arange.html#numpy.arange "numpy.arange")
(1, 8))

[plt.show](https://matplotlib.org/stable/api/_as_gen/matplotlib.pyplot.show.html#matplotlib.pyplot.show "matplotlib.pyplot.show")
()
