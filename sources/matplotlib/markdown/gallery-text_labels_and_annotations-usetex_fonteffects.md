Usetex font effects[#](https://matplotlib.org/stable/gallery/text_labels_and_annotations/usetex_fonteffects.html#usetex-font-effects "Link to this heading")

=============================================================================================================================================================

This script demonstrates that font effects specified in your pdftex.map are now supported in usetex mode.

![Usetex font effects](https://matplotlib.org/stable/_images/sphx_glr_usetex_fonteffects_001_2_00x.png)

import matplotlib.pyplot as plt

def setfont([font](https://docs.python.org/3/library/stdtypes.html#str "builtins.str")
):
    return rf'\\font\\a {[font](https://docs.python.org/3/library/stdtypes.html#str "builtins.str")
} at 14pt\\a '

[fig](https://matplotlib.org/stable/api/_as_gen/matplotlib.figure.Figure.html#matplotlib.figure.Figure "matplotlib.figure.Figure")
 \= [plt.figure](https://matplotlib.org/stable/api/_as_gen/matplotlib.pyplot.figure.html#matplotlib.pyplot.figure "matplotlib.pyplot.figure")
()
for [y](https://docs.python.org/3/library/functions.html#int "builtins.int")
, [font](https://docs.python.org/3/library/stdtypes.html#str "builtins.str")
, [text](https://docs.python.org/3/library/stdtypes.html#str "builtins.str")
 in zip(
    range(5),
    \['ptmr8r', 'ptmri8r', 'ptmro8r', 'ptmr8rn', 'ptmrr8re'\],
    \[f'Nimbus Roman No9 L {x}'\
     for x in \['', 'Italics (real italics for comparison)',\
               '(slanted)', '(condensed)', '(extended)'\]\],
):
    [fig.text](https://matplotlib.org/stable/api/_as_gen/matplotlib.figure.Figure.text.html#matplotlib.figure.Figure.text "matplotlib.figure.Figure.text")
(.1, 1 \- ([y](https://docs.python.org/3/library/functions.html#int "builtins.int")
 + 1) / 6, setfont([font](https://docs.python.org/3/library/stdtypes.html#str "builtins.str")
) + [text](https://docs.python.org/3/library/stdtypes.html#str "builtins.str")
, usetex\=True)

[fig.suptitle](https://matplotlib.org/stable/api/_as_gen/matplotlib.figure.Figure.suptitle.html#matplotlib.figure.Figure.suptitle "matplotlib.figure.Figure.suptitle")
('Usetex font effects')
\# Would also work if saving to pdf.
[plt.show](https://matplotlib.org/stable/api/_as_gen/matplotlib.pyplot.show.html#matplotlib.pyplot.show "matplotlib.pyplot.show")
()
