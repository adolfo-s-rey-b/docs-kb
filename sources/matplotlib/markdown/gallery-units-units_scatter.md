The example below shows support for unit conversions over masked arrays.

This example requires [`basic_units.py`](https://matplotlib.org/stable/_downloads/0f0fd288c7d4a6a16f4835b96343f597/basic_units.py)

![units scatter](https://matplotlib.org/stable/_images/sphx_glr_units_scatter_001_2_00x.png)

from basic\_units import hertz, minutes, secs

import matplotlib.pyplot as plt
import numpy as np

\# create masked array
[data](https://docs.python.org/3/library/stdtypes.html#tuple "builtins.tuple")
 \= (1, 2, 3, 4, 5, 6, 7, 8)
[mask](https://docs.python.org/3/library/stdtypes.html#tuple "builtins.tuple")
 \= (1, 0, 1, 0, 0, 0, 1, 0)
xsecs \= secs \* [np.ma.MaskedArray](https://numpy.org/doc/stable/reference/maskedarray.baseclass.html#numpy.ma.MaskedArray "numpy.ma.MaskedArray")
([data](https://docs.python.org/3/library/stdtypes.html#tuple "builtins.tuple")
, [mask](https://docs.python.org/3/library/stdtypes.html#tuple "builtins.tuple")
, float)

[fig](https://matplotlib.org/stable/api/_as_gen/matplotlib.figure.Figure.html#matplotlib.figure.Figure "matplotlib.figure.Figure")
, ([ax1](https://matplotlib.org/stable/api/_as_gen/matplotlib.axes.Axes.html#matplotlib.axes.Axes "matplotlib.axes.Axes")
, [ax2](https://matplotlib.org/stable/api/_as_gen/matplotlib.axes.Axes.html#matplotlib.axes.Axes "matplotlib.axes.Axes")
, [ax3](https://matplotlib.org/stable/api/_as_gen/matplotlib.axes.Axes.html#matplotlib.axes.Axes "matplotlib.axes.Axes")
) \= [plt.subplots](https://matplotlib.org/stable/api/_as_gen/matplotlib.pyplot.subplots.html#matplotlib.pyplot.subplots "matplotlib.pyplot.subplots")
(nrows\=3, sharex\=True)

[ax1.scatter](https://matplotlib.org/stable/api/_as_gen/matplotlib.axes.Axes.scatter.html#matplotlib.axes.Axes.scatter "matplotlib.axes.Axes.scatter")
(xsecs, xsecs)
[ax1.yaxis.set\_units](https://matplotlib.org/stable/api/_as_gen/matplotlib.axis.Axis.set_units.html#matplotlib.axis.Axis.set_units "matplotlib.axis.Axis.set_units")
(secs)
[ax2.scatter](https://matplotlib.org/stable/api/_as_gen/matplotlib.axes.Axes.scatter.html#matplotlib.axes.Axes.scatter "matplotlib.axes.Axes.scatter")
(xsecs, xsecs, yunits\=hertz)
[ax3.scatter](https://matplotlib.org/stable/api/_as_gen/matplotlib.axes.Axes.scatter.html#matplotlib.axes.Axes.scatter "matplotlib.axes.Axes.scatter")
(xsecs, xsecs, yunits\=minutes)

[fig.tight\_layout](https://matplotlib.org/stable/api/_as_gen/matplotlib.figure.Figure.tight_layout.html#matplotlib.figure.Figure.tight_layout "matplotlib.figure.Figure.tight_layout")
()
[plt.show](https://matplotlib.org/stable/api/_as_gen/matplotlib.pyplot.show.html#matplotlib.pyplot.show "matplotlib.pyplot.show")
()
